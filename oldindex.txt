    <script>
      (function () {
        var track = document.querySelector(".carousel-track");
        if (!track) return;

        // helper to get item width + gap
        function itemWidth() {
          var item = track.querySelector(".carousel-item");
          if (!item) return 220;
          var gap = parseInt(getComputedStyle(track).gap) || 12;
          return Math.round(item.getBoundingClientRect().width + gap);
        }

        // prevent native image drag
        track.querySelectorAll("img").forEach(function (img) {
          img.draggable = false;
        });

        // clone slides to allow smooth infinite looping
        var originalSlides = Array.from(track.children);
        var originalCount = originalSlides.length;

        // append clones of originals
        originalSlides.forEach(function (s) {
          track.appendChild(s.cloneNode(true));
        });

        // prepend clones of originals (in reverse order)
        for (var i = originalCount - 1; i >= 0; i--) {
          track.insertBefore(
            originalSlides[i].cloneNode(true),
            track.firstChild
          );
        }

        // start positioned at the first original slide
        function jumpToStart() {
          track.scrollLeft = itemWidth() * originalCount;
        }
        // allow layout to settle
        setTimeout(jumpToStart, 50);

        var prev = document.querySelector(".carousel-arrow.prev");
        var next = document.querySelector(".carousel-arrow.next");

        // prevent mouse/touch/pointer interactions from focusing the arrow buttons
        function preventPointerFocus(btn) {
          if (!btn) return;
          btn.addEventListener("pointerdown", function (e) {
            if (
              e.pointerType === "mouse" ||
              e.pointerType === "pen" ||
              e.pointerType === "touch"
            ) {
              e.preventDefault();
            }
          });
        }
        preventPointerFocus(prev);
        preventPointerFocus(next);

        // index-based smooth scrolling for less staccato feel
        var currentIndex = 0; // 0..originalCount-1
        function normalizeIndex(i) {
          return ((i % originalCount) + originalCount) % originalCount;
        }

        function scrollToIndex(i, behavior) {
          var iw = itemWidth();
          var target = (i + originalCount) * iw; // account for prepended clones
          track.scrollTo({
            left: Math.round(target),
            behavior: behavior || "smooth",
          });
        }

        // initial position
        currentIndex = 0;
        setTimeout(function () {
          scrollToIndex(currentIndex, "auto");
        }, 60);

        prev &&
          prev.addEventListener("click", function () {
            currentIndex = normalizeIndex(currentIndex - 1);
            scrollToIndex(currentIndex, "smooth");
          });
        next &&
          next.addEventListener("click", function () {
            currentIndex = normalizeIndex(currentIndex + 1);
            scrollToIndex(currentIndex, "smooth");
          });

        // pointer drag for mouse & touch
        var isDown = false;
        var startX = 0;
        var scrollLeft = 0;
        track.tabIndex = 0;

        track.addEventListener("pointerdown", function (e) {
          isDown = true;
          track.classList.add("dragging");
          startX = e.clientX;
          scrollLeft = track.scrollLeft;
          try {
            track.setPointerCapture(e.pointerId);
          } catch (err) {}
        });
        track.addEventListener("pointermove", function (e) {
          if (!isDown) return;
          var x = e.clientX;
          var walk = startX - x;
          track.scrollLeft = scrollLeft + walk;
        });
        track.addEventListener("pointerup", function (e) {
          isDown = false;
          track.classList.remove("dragging");
          try {
            track.releasePointerCapture(e.pointerId);
          } catch (err) {}
        });
        track.addEventListener("pointerleave", function () {
          isDown = false;
          track.classList.remove("dragging");
        });

        // keyboard support
        track.addEventListener("keydown", function (e) {
          if (e.key === "ArrowRight") {
            next && next.click();
          }
          if (e.key === "ArrowLeft") {
            prev && prev.click();
          }
        });

        // update currentIndex on scroll, and handle clone-region adjustments after scrolling stops
        var scrollEndTimeout = null;
        track.addEventListener("scroll", function () {
          // update current index roughly
          var iw = itemWidth();
          var rawIndex = Math.round(track.scrollLeft / iw) - originalCount;
          currentIndex = normalizeIndex(rawIndex);

          // debounce scroll end
          if (scrollEndTimeout) clearTimeout(scrollEndTimeout);
          scrollEndTimeout = setTimeout(function () {
            // ensure we're snapped exactly to the index
            scrollToIndex(currentIndex, "smooth");

            // if we're in cloned region (due to momentum), jump invisibly to original
            var left = track.scrollLeft;
            var start = iw * originalCount;
            var end = iw * (originalCount * 2);
            if (left < start - iw / 2) {
              // jumped into prepended clones
              track.scrollLeft = left + iw * originalCount;
            } else if (left > end + iw / 2) {
              // jumped into appended clones
              track.scrollLeft = left - iw * originalCount;
            }
          }, 120);
        });

        // hide arrows if no overflow (consider clones)
        function updateArrows() {
          if (!prev || !next) return;
          if (track.scrollWidth <= track.clientWidth + 1) {
            prev.style.display = "none";
            next.style.display = "none";
          } else {
            prev.style.display = "flex";
            next.style.display = "flex";
          }
        }

        window.addEventListener("resize", updateArrows);
        updateArrows();
      })();
    </script>
